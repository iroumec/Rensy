// Viewport transform.
export inline Vector3D projectVector(Vector3D vector, unsigned width, unsigned height)
{
    // First, ortographics projection -> Z axis is discard.
    // Second, since the input models are scaled to have fir in the [-1.1]^3 world coordinates,
    // the vector is scaled to span the entire screen.

    // First, the interval is moved from [-1, 1]^3 to [0, 2]^3 by adding 1 to each coordinate.
    // Second, the intervals are multiplied by width or height or max colour, the invervals are ([0, 2W], [0, 2H], [0, 510]).
    // Finally, the intervals are divided by 2:  ([0, W], [0, H], [0, 255]).
    Vector3D out;
    out.x() = (vector.x() + 1.) * width / 2;
    out.y() = (vector.y() + 1.) * height / 2;
    out.z() = (vector.z() + 1.) * 255. / 2; // Here we save the colours.
    // "Discarding" or giving another meaning to the z coordinate is part of the ortographic projection.

    return out;
}
